    async function processAI() {
        if (!checkAuth()) return;

        if (!isVip && credits <= 0) {
            openModal('modal-vip');
            return;
        }

        const uiLang = document.getElementById('uiLang').value;
        const t = translations[uiLang];

        const key = document.getElementById('apiKey').value.trim();
        const text = srcText.value.trim();
        const task = document.getElementById('task').value;
        const targetLang = document.getElementById('targetLang').value;
        const out = document.getElementById('output');

        if (!key) return alert(t.keyAlert);
        if (!text) return alert(t.textAlert);

        out.innerText = t.processing;

        let prompt = `Task: ${task}. Target Language: ${targetLang}.\nText: ${text}`;
        if(task === 'fix') prompt = `Correct grammar and style of this text. Respond in ${targetLang}: ${text}`;
        if(task === 'rephrase') prompt = `Rewrite this professionally in ${targetLang}: ${text}`;
        if(task === 'summarize') prompt = `Summarize this text in 2-3 sentences in ${targetLang}: ${text}`;
        if(task === 'translate') prompt = `Translate this text directly to ${targetLang}: ${text}`;

        try {
            const res = await fetch('https://api.openai.com/v1/chat/completions', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'Authorization': `Bearer ${key}`
                },
                body: JSON.stringify({
                    model: "gpt-3.5-turbo",
                    messages: [{ role: "user", content: prompt }],
                    temperature: 0.3
                })
            });

            const data = await res.json();
            if (data.error) {
                out.innerText = 'Error: ' + data.error.message;
            } else {
                out.innerText = data.choices[0].message.content.trim();
                if (!isVip) {
                    credits--;
                    updateCreditUI();
                }
            }
        } catch (e) {
            out.innerText = 'Error processing request. Check your API key and connection.';
        }
